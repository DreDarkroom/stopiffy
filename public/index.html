Pick a track—

    Use it in your game

    <script src="https://YOUR-HOST/sdk/stopiffy.js"></script>
<script>
  const sp = new Stopiffy.Client({ base: 'https://YOUR-HOST' });
  const ctx = new AudioContext();
  const tracks = await sp.tracks();
  const buf = await sp.load(tracks[0]);   // AudioBuffer, loops seamlessly
  const src = ctx.createBufferSource();
  src.buffer = buf; src.loop = true; src.connect(ctx.destination); src.start();
</script>
